cmake_minimum_required(VERSION 3.20)
project(duck_gym VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DUCK_SANITIZE "Enable selected sanitizers" OFF)
set(DUCK_SANITIZERS "address,undefined" CACHE STRING "Comma-separated sanitizer list")
add_library(duck_cpu src/solver.cpp)
target_include_directories(duck_cpu PUBLIC include)
target_compile_options(duck_cpu PRIVATE -Wall -Wextra -Wpedantic)
if(DUCK_SANITIZE)
  target_compile_options(duck_cpu PUBLIC -fsanitize=${DUCK_SANITIZERS} -fno-omit-frame-pointer)
  target_link_options(duck_cpu PUBLIC -fsanitize=${DUCK_SANITIZERS})
endif()
add_executable(duck_sim src/main.cpp)
target_link_libraries(duck_sim PRIVATE duck_cpu)
enable_testing()
add_executable(duck_math_test tests/math_test.cpp)
target_link_libraries(duck_math_test PRIVATE duck_cpu)
add_test(NAME math COMMAND duck_math_test)

add_executable(duck_state_test tests/state_test.cpp)
target_link_libraries(duck_state_test PRIVATE duck_cpu)
add_test(NAME state COMMAND duck_state_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/pendulum.duck)
set_tests_properties(math state PROPERTIES TIMEOUT 30)

option(DUCK_BUILD_PYTHON "Build pybind11 CPU batch extension" OFF)
if(DUCK_BUILD_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  set_target_properties(duck_cpu PROPERTIES POSITION_INDEPENDENT_CODE ON)
  pybind11_add_module(_duck_reference src/cuda/cpu_bindings.cpp)
  target_link_libraries(_duck_reference PRIVATE duck_cpu Threads::Threads)
  target_compile_options(_duck_reference PRIVATE -Wno-narrowing -ffp-contract=off)
  set_target_properties(_duck_reference PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
  pybind11_add_module(_duck_cpu src/bindings.cpp)
  target_link_libraries(_duck_cpu PRIVATE duck_cpu Threads::Threads)
  set_target_properties(_duck_cpu PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
endif()
